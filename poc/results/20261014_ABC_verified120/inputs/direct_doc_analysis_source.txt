1345:    async def _try_direct_doc_analysis(
1346:        self,
1347:        query: str,
1348:        paths: List[str],
1349:    ) -> Optional[str]:
1350:        """Short-circuit for document-level queries (e.g. "请总结这篇文档").
1351:
1352:        Uses the LLM to classify query intent (language-agnostic).  When
1353:        a whole-document operation is detected **and** suitable files exist
1354:        in *paths*, their content is fed directly to the LLM — bypassing
1355:        the heavyweight keyword / dir-scan / evidence pipeline.
1356:
1357:        Returns:
1358:            LLM answer string, or None if the short-circuit does not apply.
1359:        """
1360:        from sirchmunk.doc_qa import (
1361:            detect_doc_intent,
1362:            collect_doc_files,
1363:            analyse_documents,
1364:        )
1365:
1366:        # Step 1: file gate — skip early if paths contain no loadable docs
1367:        doc_files = collect_doc_files(paths)
1368:        if not doc_files:
1369:            return None
1370:
1371:        # Step 2: LLM intent classification (cheap, stream=False)
1372:        operation = await detect_doc_intent(query, self.llm, self.llm_usages)
1373:        if operation is None:
1374:            return None
1375:
1376:        filenames = ", ".join(Path(d.path).name for d in doc_files)
1377:        await self._logger.info(
1378:            f"[DocQA] Intent '{operation}' detected — "
1379:            f"loading {len(doc_files)} file(s) for direct analysis: {filenames}"
1380:        )
1381:
1382:        # Step 3: for summary operations, use the chunked summarizer
1383:        # with optional smart dir scanning; for other operations, use the
1384:        # general analyser.
1385:        if operation in ("summarize", "summary", "extract"):
1386:            scan_result = None
1387:            if self._has_directory_paths(paths):
1388:                scan_result = await self._probe_dir_scan(paths, max_files=300)
1389:            answer = await self._summarize_documents(
1390:                query, paths, scan_result=scan_result,
1391:            )
1392:        else:
1393:            answer = await analyse_documents(
1394:                query=query,
1395:                doc_files=doc_files,
1396:                llm=self.llm,
1397:                llm_usages=self.llm_usages,
1398:            )
1399:
1400:        if answer:
1401:            await self._logger.success("[DocQA] Direct document analysis complete")
1402:        return answer
