62:async def detect_doc_intent(
63:    query: str,
64:    llm: OpenAIChat,
65:    llm_usages: Optional[List[Dict[str, Any]]] = None,
66:) -> Optional[str]:
67:    """Use the LLM to decide whether *query* is a whole-document operation.
68:
69:    Returns the operation name (e.g. ``"summarize"``, ``"translate"``)
70:    when the LLM classifies the query as document-level, or ``None``
71:    when it is a normal search/retrieval query.
72:
73:    This is a lightweight call (short prompt, ``stream=False``) so it
74:    adds minimal latency as a pre-gate.
75:    """
76:    prompt = DETECT_DOC_INTENT.format(user_input=query)
77:    try:
78:        response = await llm.achat(
79:            messages=[{"role": "user", "content": prompt}],
80:            stream=False,
81:        )
82:        if llm_usages is not None:
83:            llm_usages.append(response.usage)
84:
85:        parsed = _parse_json_response(response.content)
86:        if parsed and parsed.get("doc_level") is True:
87:            return parsed.get("op", "analyze")
88:    except Exception as exc:
89:        logger.debug(f"[DocQA] Intent detection failed, falling back: {exc}")
90:
91:    return None

94:def collect_doc_files(
95:    paths: List[str],
96:    max_file_size: int = DIRECT_LOAD_MAX_FILE_SIZE,
97:    max_files: int = MAX_DOC_FILES,
98:) -> List[DocFile]:
99:    """Collect analyzable document files from *paths*.
100:
101:    For each path element:
102:      - If it is a file with a supported extension → include it.
103:      - If it is a directory → include immediate children with supported
104:        extensions (non-recursive to avoid scanning large trees).
105:
106:    Args:
107:        paths: Resolved search path strings.
108:        max_file_size: Skip files larger than this (bytes).
109:        max_files: Maximum number of files to collect.
110:
111:    Returns:
112:        List of DocFile descriptors, sorted by size ascending.
113:    """
114:    collected: List[DocFile] = []
115:    seen: set = set()
116:
117:    for p_str in paths:
118:        p = Path(p_str)
119:        if not p.exists():
120:            continue
121:
122:        entries = [p] if p.is_file() else _list_dir_files(p)
123:
124:        for fp in entries:
125:            ext = fp.suffix.lower()
126:            if ext not in _DOC_EXTENSIONS:
127:                continue
128:
129:            resolved = str(fp.resolve())
130:            if resolved in seen:
131:                continue
132:            seen.add(resolved)
133:
134:            try:
135:                size = fp.stat().st_size
136:            except OSError:
137:                continue
138:
139:            if size == 0 or size > max_file_size:
140:                if size > max_file_size:
141:                    logger.debug(
142:                        f"[DocQA] Skipping {fp.name}: "
143:                        f"{size / 1024 / 1024:.1f} MB exceeds limit "
144:                        f"({max_file_size / 1024 / 1024:.0f} MB)"
145:                    )
146:                continue
147:
148:            collected.append(DocFile(
149:                path=resolved,
150:                size=size,
151:                extension=ext,
152:                is_text=ext in _TEXT_EXTENSIONS,
153:            ))
154:            if len(collected) >= max_files:
155:                break
156:
157:        if len(collected) >= max_files:
158:            break
159:
160:    collected.sort(key=lambda d: d.size)
161:    return collected
